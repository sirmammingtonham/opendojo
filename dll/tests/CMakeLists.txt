cmake_minimum_required(VERSION 3.20)
project(OpenDojoReliabilityTests LANGUAGES C CXX)
enable_testing()
add_executable(reliability_tests reliability_tests.cpp ../src/memory.cpp)
target_include_directories(reliability_tests PRIVATE ../include)
target_compile_features(reliability_tests PRIVATE cxx_std_20)
target_compile_options(reliability_tests PRIVATE /W4 /utf-8)
add_test(NAME reliability COMMAND reliability_tests)

add_executable(runtime_tests runtime_tests.cpp ../src/memory.cpp ../src/file_io.cpp)
add_executable(import_tests import_tests.cpp ../src/slot.cpp ../src/memory.cpp)
add_executable(session_tests session_tests.cpp ../src/subsystems.cpp ../src/memory.cpp)
foreach(test_target runtime_tests import_tests session_tests)
    target_include_directories(${test_target} PRIVATE ../include)
    target_compile_features(${test_target} PRIVATE cxx_std_20)
    target_compile_options(${test_target} PRIVATE /W4 /utf-8)
    add_test(NAME ${test_target} COMMAND ${test_target})
    set_tests_properties(${test_target} PROPERTIES TIMEOUT 30)
endforeach()

find_path(OPENDOJO_TEST_JSON_INCLUDE nlohmann/json.hpp
    HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../build/_deps/nlohmann_json-src/single_include"
    REQUIRED)
add_executable(config_tests config_tests.cpp ../src/config.cpp ../src/file_io.cpp)
target_include_directories(config_tests PRIVATE ../include "${OPENDOJO_TEST_JSON_INCLUDE}")
target_compile_features(config_tests PRIVATE cxx_std_20)
target_compile_options(config_tests PRIVATE /W4 /utf-8)
foreach(scenario legacy embedded settings types)
    add_test(NAME config_${scenario} COMMAND config_tests ${scenario})
    set_tests_properties(config_${scenario} PROPERTIES TIMEOUT 15)
endforeach()

set(OPENDOJO_TEST_DLL "" CACHE FILEPATH "Optional built proxy DLL for a separate-process lifetime test")

    add_test(NAME proxy_lifetime COMMAND proxy_lifetime_test "${OPENDOJO_TEST_DLL}")
    set_tests_properties(proxy_lifetime PROPERTIES TIMEOUT 15)
endif()

        set_tests_properties(native_layout_${scenario} PROPERTIES TIMEOUT 60)
    endforeach()
endif()
